# Releasing a new version

# sanity check
  make rcheck
  
# update the version pyproject.toml
  git commit pyproject.toml -m 'update version' 

# update CHANGELOG.rst  `git shortlog v0.9.3..HEAD`
  git commit -m 'update recent changes' CHANGELOG.rst
  git push

# deploy JupyterLite files
  make realclean
  make lite
  make lite-deploy

# create release tag on github
  git tag 1.0.0
  git push origin 1.0.0

# manually publish release at
  open https://github.com/scottprahl/pypolar

# verify github action success
  open https://pypi.org/project/pypolar/
  open https://pypolar.readthedocs.io
  open https://doi.org/10.5281/zenodo.8358111

# wait for feedstock update
  open https://github.com/conda-forge/pypolar-feedstock
